use std::process::Command;
use std::path::Path;
fn windows_file_open_command(path: &Path) -> std::io::Result<Command> {
    // Explorer resolves the file association without interpreting the path as
    // a cmd.exe program (including & operators and %variable% expansion).
    let mut command = Command::new("explorer.exe");
    // Callers may supply relative paths. Resolve against Usher's working
    // directory instead of leaving that interpretation to Explorer.
    command.arg(std::path::absolute(path)?);
    Ok(command)
}
fn executable_restart_command(
    executable: &std::path::Path,
    args: &[std::ffi::OsString],
) -> std::process::Command {
    // Keep executable paths and arguments out of a command interpreter on
    // every platform, including Windows paths containing shell metacharacters.
    let mut command = std::process::Command::new(executable);
    command.args(args);
    command
}
fn describe(label: &str, command: &Command) {
    println!("{label}: program={:?}; args={:?}", command.get_program(), command.get_args().collect::<Vec<_>>());
}
fn main() {
    let path = Path::new("fixture/reports/notes & literal.md");
    let executable = Path::new("fixture/application dir/usher & literal.exe");
    let args: Vec<std::ffi::OsString> = vec!["literal argument".into()];
    let mut before_file = Command::new("cmd");
    before_file.args(["/C", "start", ""]).arg(path);
    describe("HEAD Windows file open", &before_file);
    let after_file = windows_file_open_command(path).unwrap();
    describe("candidate Windows file open", &after_file);
    assert_eq!(before_file.get_program(), "cmd");
    assert_eq!(after_file.get_program(), "explorer.exe");
    assert_eq!(after_file.get_args().count(), 1);
    assert!(Path::new(after_file.get_args().next().unwrap()).is_absolute());
    let mut before_restart = Command::new("cmd");
    before_restart.args(["/c", "start", "", &executable.to_string_lossy()]).args(&args);
    describe("HEAD Windows restart", &before_restart);
    let after_restart = executable_restart_command(executable, &args);
    describe("candidate Windows restart", &after_restart);
    assert_eq!(after_restart.get_program(), executable.as_os_str());
    assert_eq!(after_restart.get_args().collect::<Vec<_>>(), args);
    println!("PASS: source-verified command construction comparison; no command was spawned");
}
